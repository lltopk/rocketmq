            if (null != context) {

            inFlightRequestMap.remove(responseCommand.getOpaque());
        }
        return super.writeAndFlush(msg);
    }


    @Override
    public void registerInvocationContext(int opaque, InvocationContextInterface context) {
        inFlightRequestMap.put(opaque, context);
    }

        this.response = resp;

        return System.currentTimeMillis() - timestamp >= Duration.ofSeconds(expiredTimeSec).toMillis();
    }

    public CompletableFuture<RemotingCommand> getResponse() {
        return response;
    }

    //省略该类其他内容

    @Override
    public void ackMessage(AckMessageRequest request, StreamObserver<AckMessageResponse> responseObserver) {

 * @课程描述:从零带你写框架系列中的课程，整个系列包含netty，xxl-job，rocketmq，nacos，sofajraft，spring，springboot，disruptor，编译器，虚拟机等等。

public class PopBufferMergeService extends ServiceThread {

    private static final Logger POP_LOGGER = LoggerFactory.getLogger(LoggerName.ROCKETMQ_POP_LOGGER_NAME);

    //存储待确认的PopCheckPointWrapper消息的缓冲区，key是这批消息的唯一表示
    //注意，每一PopCheckPointWrapper对象封装的都是一个PopCheckPoint对象，而每一个PopCheckPoint对象对应的都是一批待ack的消息，因为消息是一批一批从队列中弹出来的
    //key=point.getTopic() + point.getCId() + point.getQueueId() + point.getStartOffset() + point.getPopTime() + point.getBrokerName();
    ConcurrentHashMap<String/*mergeKey*/, PopCheckPointWrapper> buffer = new ConcurrentHashMap<>(1024 * 16);

    //服务是否可用的标志，默认是可用的
    private volatile boolean serving = true;

    private final BrokerController brokerController;

    //省略该类其他内容





    /**
     * @课程描述:从零带你写框架系列中的课程，整个系列包含netty，xxl-job，rocketmq，nacos，sofajraft，spring，springboot，disruptor，编译器，虚拟机等等。
     * @author：陈清风扬，个人微信号：chenqingfengyangjj。
     * @方法描述：ack消息的方法
     */
    public boolean addAk(int reviveQid, AckMsg ackMsg) {
        //判断当前Broker节点是否启用了当前组件功能
        if (!brokerController.getBrokerConfig().isEnablePopBufferMerge()) {
            return false;
        }
        //判断服务是否能正常工作
        if (!serving) {
            return false;
        }

        
        try {
            //根据ack消息得到消息唯一key，然后从buffer得到待确认的消息
            PopCheckPointWrapper pointWrapper = this.buffer.get(ackMsg.getTopic() + ackMsg.getConsumerGroup() + ackMsg.getQueueId() + ackMsg.getStartOffset() + ackMsg.getPopTime() + ackMsg.getBrokerName());
            
            //没找到对应的ck消息包装对象直接返回false
            if (pointWrapper == null) {
                if (brokerController.getBrokerConfig().isEnablePopLog()) {
                    POP_LOGGER.warn("[PopBuffer]add ack fail, rqId={}, no ck, {}", reviveQid, ackMsg);
                }
                return false;
            }
            
          
            //得到真正的ck检查点对象
            PopCheckPoint point = pointWrapper.getCk();
            //得到当前系统时间
            long now = System.currentTimeMillis();


            //判断当前要处理的ck消息是否已经要超时了，如果快超时了，则默认已经不够处理了，直接返回false
            if (point.getReviveTime() - now < brokerController.getBrokerConfig().getPopCkStayBufferTimeOut() + 1500) {
                if (brokerController.getBrokerConfig().isEnablePopLog()) {
                    POP_LOGGER.warn("[PopBuffer]add ack fail, rqId={}, almost timeout for revive, {}, {}, {}", reviveQid, pointWrapper, ackMsg, now);
                }
                return false;
            }

            
            //判断待确认的消息是否在内存中停留的时间过长，如果太长也默认不处理，视为过期ck消息，直接返回false
            if (now - point.getPopTime() > brokerController.getBrokerConfig().getPopCkStayBufferTime() - 1500) {
                if (brokerController.getBrokerConfig().isEnablePopLog()) {
                    POP_LOGGER.warn("[PopBuffer]add ack fail, rqId={}, stay too long, {}, {}, {}", reviveQid, pointWrapper, ackMsg, now);
                }
                return false;
            }

            //判断ack消息是否为批量确认消息，在我们的程序中并没有使用批量确认功能，rocketmq源码中默认也没有启用这个功能
            //所以下面这个分支大家忽略即可
            if (ackMsg instanceof BatchAckMsg) {

                //省略这部分内容
                
            } else {
                
                //执行到这里就是确认单条消息的操作，这里得到要ack的消息在ck封装的这批要确认的消息的位置
                int indexOfAck = point.indexOfAck(ackMsg.getAckOffset());
                //判断索引是否有效
                if (indexOfAck > -1) {
                    //索引有效则更新该位置的位图值，设置为已确认状态
                    markBitCAS(pointWrapper.getBits(), indexOfAck);
                    
                } else {
                    //索引无效则记录错误日志
                    POP_LOGGER.error("[PopBuffer]Invalid index of ack, reviveQid={}, {}, {}", reviveQid, ackMsg, point);
                    //返回true，表示消息ack成功
                    return true;
                }
            }

            if (brokerController.getBrokerConfig().isEnablePopLog()) {
                POP_LOGGER.info("[PopBuffer]add ack, rqId={}, {}, {}", reviveQid, pointWrapper, ackMsg);
            }
            return true;
        } catch (Throwable e) {
            POP_LOGGER.error("[PopBuffer]add ack error, rqId=" + reviveQid + ", " + ackMsg, e);
        }
        return false;
    }




    /**
     * @课程描述:从零带你写框架系列中的课程，整个系列包含netty，xxl-job，rocketmq，nacos，sofajraft，spring，springboot，disruptor，编译器，虚拟机等等。
     * @author：陈清风扬，个人微信号：chenqingfengyangjj。
     * @方法描述：原子操作设置消息被确认位标志的方法
     */
    private void markBitCAS(AtomicInteger setBits, int index) {
        while (true) {
            //得到消息被确认位图
            int bits = setBits.get();
            //得到位图指定位置的消息，并判断该消息是否已被确认
            if (DataConverter.getBit(bits, index)) {
                //如果已被确认则直接退出
                break;
            }
            //得到消息被确认后的新值，其实是位图值发生了变化，得到了新的int值
            int newBits = DataConverter.setBit(bits, index, true);
            //cas更新消息被确认的位图标志，也就是设置AtomicInteger的新值
            if (setBits.compareAndSet(bits, newBits)) {
                break;
            }
        }
    }


    //省略该类其他内容

}
