```java
/**
 * @课程描述:从零带你写框架系列中的课程，整个系列包含netty，xxl-job，rocketmq，nacos，sofajraft，spring，springboot，disruptor，编译器，虚拟机等等。

@SuppressWarnings({"UnstableApiUsage", "NullableProblems"})
public abstract class ClientImpl extends AbstractIdleService implements Client, ClientSessionHandler, MessageInterceptor {

    //客户端管理器
    protected final ClientManager clientManager;

    //客户端配置信息对象
    protected final ClientConfiguration clientConfiguration;

    //客户端订阅的主题集合
    protected final Set<String> topics;

    //客户端缓存主题路由信息的map表
    private final ConcurrentMap<String, TopicRouteData> topicRouteCache;

    //存储客户端和不服务端建立的会话的map表
    @GuardedBy("sessionsLock")
    private final Map<Endpoints, ClientSessionImpl> sessionsTable;

    //省略该类的其他内容


    /**
     * @课程描述:从零带你写框架系列中的课程，整个系列包含netty，xxl-job，rocketmq，nacos，sofajraft，spring，springboot，disruptor，编译器，虚拟机等等。
     * @author：陈清风扬，个人微信号：chenqingfengyangjj。
     * @方法描述：启动客户端的方法
     */
    @Override
    protected void startUp() throws Exception {
        LOGGER.info("Begin to start the rocketmq client, clientId={}", clientId);
        //在这里启动了客户端管理器，并且等待客户端管理器运行成功，然后线程才继续向下执行
        //这里的操作就会把客户端管理器中的各个定时任务提交给定时任务执行器
        this.clientManager.startAsync().awaitRunning();
        LOGGER.info("Begin to fetch topic(s) route data from remote during client startup, clientId={}, topics={}",
                clientId, topics);
        //这里执行了fetchTopicRoute方法，也就是说在客户端启动的过程中，就已经从对应的服务端获取了对应主题的路由信息
        final List<ListenableFuture<TopicRouteData>> futures =
                topics.stream().map(this::fetchTopicRoute).collect(Collectors.toList());
        //等待获取所有主题路由信息操作完成，然后当前线程才能继续向下执行
        for (ListenableFuture<TopicRouteData> future : futures) {
            future.get();
        }
        LOGGER.info("Fetch topic route data from remote successfully during startup, clientId={}, topics={}",
                clientId, topics);
        //从客户端管理器中得到定时任务执行器
        final ScheduledExecutorService scheduler = clientManager.getScheduler();
        //在这里向定时任务执行器提交定期更新缓存的主题路由信息的任务
        this.updateRouteCacheFuture = scheduler.scheduleWithFixedDelay(() -> {
            try {
                //更新缓存的主题路由信息的方法
                updateRouteCache();
            } catch (Throwable t) {
                LOGGER.error("Exception raised while updating topic route cache, clientId={}", clientId, t);
            }
        }, 10, 30, TimeUnit.SECONDS);
        LOGGER.info("The rocketmq client starts successfully, clientId={}", clientId);
    }
```

```java
package org.apache.rocketmq.client.java.impl;


/**
 * @课程描述:从零带你写框架系列中的课程，整个系列包含netty，xxl-job，rocketmq，nacos，sofajraft，spring，springboot，disruptor，编译器，虚拟机等等。
 * @author：陈清风扬，个人微信号：chenqingfengyangjj。
 * @date:2025/6/10
 * @方法描述：客户端管理器实现类
 */
public class ClientManagerImpl extends ClientManager {

    //客户端对象
    private final Client client;

    //RPC客户端连接表，key为服务端地址，value为对应的RPC客户端连接
    @GuardedBy("rpcClientTableLock")
    private final Map<Endpoints, RpcClient> rpcClientTable;

    //保护客户端连接表并发安全的读写锁
    private final ReadWriteLock rpcClientTableLock;

    //定时任务执行器
    private final ScheduledExecutorService scheduler;

    //异步工作执行器，这个执行器专门执行rpc客户端发送请求的操作
    private final ExecutorService asyncWorker;

    //省略该类其他内容

    /**
     * @课程描述:从零带你写框架系列中的课程，整个系列包含netty，xxl-job，rocketmq，nacos，sofajraft，spring，springboot，disruptor，编译器，虚拟机等等。
     * @author：陈清风扬，个人微信号：chenqingfengyangjj。
     * @方法描述：向指定服务端发送消息的方法
     */
    @Override
    public ListenableFuture<RpcInvocation<SendMessageResponse>> sendMessage(Endpoints endpoints, Metadata metadata,
                                                                            SendMessageRequest request, Duration duration) {
        try {
            final RpcClient rpcClient = getRpcClient(endpoints);
            return rpcClient.sendMessage(metadata, request, asyncWorker, duration);
        } catch (Throwable t) {
            return Futures.immediateFailedFuture(t);
        }
    }


    /**
     * @课程描述:从零带你写框架系列中的课程，整个系列包含netty，xxl-job，rocketmq，nacos，sofajraft，spring，springboot，disruptor，编译器，虚拟机等等。
     * @author：陈清风扬，个人微信号：chenqingfengyangjj。
     * @方法描述：得到指定服务端地址对应的RPC客户端连接对象
     */
    private RpcClient getRpcClient(Endpoints endpoints) throws ClientException {
        RpcClient rpcClient;
        //得到读锁，保证并发安全
        rpcClientTableLock.readLock().lock();
        try {
            //先从客户端连接表中查找该服务端地址对应的RPC客户端连接对象
            rpcClient = rpcClientTable.get(endpoints);
            //如果找到了，就直接返回该RPC客户端连接对象
            if (null != rpcClient) {
                return rpcClient;
            }
        } finally {
            //释放读锁
            rpcClientTableLock.readLock().unlock();
        }
        //如果没有从客户端连接表中找到对应的rpc客户端对象
        //那就在这里获取写锁，因为接下来要为这个服务端地址创建一个对应的rpc客户端连接对象
        //然后把该rpc客户端对象放入客户端连接表中，获取写锁保证并发安全
        rpcClientTableLock.writeLock().lock();
        try {
            //双重判断，因为可能在获取写锁之前，其他线程已经创建了该服务端地址对应的RPC客户端连接对象
            rpcClient = rpcClientTable.get(endpoints);
            if (null != rpcClient) {
                return rpcClient;
            }
            try {
                //仍然没有找到对应的rpc客户端对象，那就在这里创建一个对应的rpc客户端对象
                rpcClient = new RpcClientImpl(endpoints);
            } catch (SSLException e) {
                LOGGER.error("Failed to get rpc client, endpoints={}", endpoints);
                throw new ClientException("Failed to generate RPC client", e);
            }
            //把新创建的rpc客户端对象放入客户端连接表中
            rpcClientTable.put(endpoints, rpcClient);
            return rpcClient;
        } finally {
            //释放写锁
            rpcClientTableLock.writeLock().unlock();
        }
    }
```

在回顾了之前的知识之后，我相信在大家的脑海里应该也形成了一个共识，那就是 ProducerImpl 生产者客户端只是一个最外层的、暴露给用户使用的对象，这个 ProducerImpl 生产者客户端似乎只是为某些操作提供了入口方法，除此之外，它并不负责更多核心工作。就比如说，用户虽然可以使用 ProducerImpl 向指定服务器发送生产消息，但是消息最终的发送还需要 ClientManagerImpl 客户端管理器以及真正发送消息的 RpcClientImpl 对象来完成；而在生产者客户端启动的过程中，我们也可以发现，客户端和服务端建立连接、从 Proxy 获得指定主题路由信息、以及执行心跳检测的操作也并不真的是由 ProducerImpl 对象来完成的，而是由 ClientImpl、ClientManagerImpl 以及 RpcClientImpl 这几个对象协作完成的。种种情况似乎都在告诉我们，虽然我们之前构建的是一个生产者客户端，但是整个客户端的核心功能和生产者并没有非常紧密的联系。



生产者客户端要做的肯定是把生产出来的消息发送给指定服务端，但说到底这无非就是客户端向服务端发送消息的操作，确切地说，无非就是客户端向服务端发送一个请求，服务端再回复给客户端一个响应而已，生产者客户端从服务端得到指定主题的路由信息、和指定服务端进行心跳检测等等活动都是在重复请求响应操作而已。如果是这样的话，那接下来就很简单了，**因为根据我们已经定义的 ClientImpl、ClientManagerImpl、RpcClientImpl 这三个类，基本上就把所谓的请求响应模式实现了，那如果我们要在此基础上实现一个消费者客户端，先不考虑消费者客户端的具体功能，只考虑客户端本身的话，那我们其实只要再为 ClientImpl 定义一个 ConsumerImpl 子类，这个子类对象不就是消费者客户端了吗**？



从功能上来说，**消费者客户端和生产者客户端有很多交集，比如说，消费者客户端和生产者客户端都需要和服务端建立网络连接，都需要在启动的过程中从 Proxy 服务端得到订阅主题的路由信息、都需要和服务端进行心跳检测活动、也都需要把自己的客户端信息注册到服务端，交给服务端来管理等等**。这些常规操作大家都可以理解吧？**而这些操作只需要使用 ClientImpl、ClientManagerImpl、RpcClientImpl 这三个类就能完成，**也就是说，我们已经把消费者客户端和生产者客户端的公共功能都提炼出来了，并且也都实现了，而在这些公共功能的基础上构建一个消费者客户端，确实是易如反掌，这一点大家应该也都认同吧？如果大家理解了以上内容，那接下来我们就可以真正构建一个消费者客户端了。

**要完善的第三个方法其实也很简单，那就是消费者客户端也要像生产者客户端那样，在和 Proxy 构建了网络连接之后，要把自己的信息都注册到 Proxy 节点，当然，这些信息最终并不一定是 Proxy 节点管理，而是 Broker 节点管理，并且还要定期和 Proxy 执行心跳检测活动，而在执行这些操作的过程中，消费者客户端还会定期从 Proxy 节点同步设置信息到本地**。对于消费者客户端来说，这些操作肯定也是要执行的吧？当然，这些操作其实已经被我们实现了，毕竟这些操作都会在 **ClientImpl、ClientManagerImpl、RpcClientImpl、ClientSessionImpl **这四个对象中执行，但有一点需要我们继续完善，那就是从 Proxy 节点同步设置信息到本地后，那些设置信息应该被保存在哪里？在生产者客户端中，设置信息都被封装在了 ProducerSettings 中(这一点我就不展开了，大家自己回顾之前的内容即可)，那来到消费者客户端中，**我们不妨就给消费者客户端定义一个 SimpleConsumerSettings 类，消费者客户端的设置信息都封装在这个类的对象中即可**。至于设置信息是什么就很简单了，无非就是消费者客户端所属消费组的名称，主题过滤信息，请求超时时间等等信息，大家可以直接去我提供的第十八版本代码中查看，我就不在文章中展示了。

```java
        //在这里创建了生产者客户端对象
        final ProducerImpl producer = new ProducerImpl(clientConfiguration, topics, maxAttempts, checker);
        //在这里启动了生产者客户端对象，异步起动，并且等待客户端启动完成
        producer.startAsync().awaitRunning();
        return producer;
    }
}
```

```java
public class SimpleConsumerBuilderImpl implements SimpleConsumerBuilder {

    private String consumerGroup = null;

    private Duration awaitDuration = null;

    public SimpleConsumerBuilder setClientConfiguration(ClientConfiguration clientConfiguration) {
        this.clientConfiguration = checkNotNull(clientConfiguration, "clientConfiguration should not be null");

    }

    public SimpleConsumerBuilder setConsumerGroup(String consumerGroup) {
        checkNotNull(consumerGroup, "consumerGroup should not be null");
        checkArgument(CONSUMER_GROUP_PATTERN.matcher(consumerGroup).matches(),
                "consumerGroup does not match the regex [regex=%s]", CONSUMER_GROUP_PATTERN.pattern());
        this.consumerGroup = consumerGroup;
        return this;
```
